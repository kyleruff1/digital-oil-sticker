# Operational endpoints are excluded from indexing. They are not secret — they
# carry no personal data and no secrets by construction — but they are not
# content, and a search result pointing at a readiness probe helps nobody.
# The endpoints also send `x-robots-tag: noindex, nofollow`, because a
# robots.txt rule is a request and a header is an instruction.

User-agent: *
Disallow: /health
Disallow: /ready
Disallow: /version

# Everything else is indexable.
Allow: /
